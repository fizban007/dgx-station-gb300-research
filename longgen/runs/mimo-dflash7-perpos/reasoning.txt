};

function initBoard() {
    board = Array(ROWS).fill(null).map(() => Array(COLS).fill(null));
}

function initGame() {
    initBoard();
    score = 0;
    level = 0;
    lines = 0;
    combo = -1;
    backToBack = false;

    paused = false;
    holdPiece = null;

    clearingRows = [];
    particles = [];
    scorePopups = [];

    nextQueue = [];
    refillQueue();
    
    spawnPiece();

function refillQueue() {
    const types = ['I','O','T','S','Z','J','L'];
    while (nextQueue.length < 5) {
        // Random with bag system (simplified - each bag of 7)
        if (types.length === 0) {
            types.push('I','O','T','S','Z','J','L');
        }
        const idx = Math.floor(Math.random() * types.length);
        nextQueue.push(types.splice(idx, 1)[0]);
    }
}

function spawnPiece() {
    const type = nextQueue.shift();
    refillQueue();
    const piece = PIECES[type];
    const boxSize = piece.boxSize;
    
    // Spawn position
    const x = boxSize === 4 ? 3 : 3;
    const y = -1;
    
    currentPiece = {
        type: type,
        state: 0,
        x: x,
        y: y,
        lastAction: null,
        lastKick: 0
    };
    
    holdUsed = false;
    lastMoveWasRotation = false;
    locking = false;
    lockTimer = 0;
    dropTimer = 0;
    
    // Check if spawn position is valid
    if (!isValidPosition(currentPiece.x, currentPiece.y, currentPiece.state)) {
        gameOver = true;
        gameOverScreen();
    }
}

// ==================== COLLISION DETECTION ====================
function isValidPosition(x, y, state) {
    const cells = PIECES[currentPiece.type].states[state];
    for (const [cx, cy] of cells) {
        const bx = x + cx;
        const by = y + cy;
        if (bx < 0 || bx >= COLS || by >= ROWS) return false;
        if (by >= 0 && board[by][bx] !== null) return false;
    }
    return true;
}

// ==================== MOVEMENT ====================
function movePiece(dx, dy) {
    const newX = currentPiece.x + dx;
    const newY = currentPiece.y + dy;
    if (isValidPosition(newX, newY, currentPiece.state)) {
        currentPiece.x = newX;
        currentPiece.y = newY;
        if (dx !== 0 || dy !== 0) {
            lastMoveWasRotation = false;
            if (locking) {
                lockTimer = 0; // Reset lock delay on successful move
            }
            // Check if we should start locking
            if (dy > 0 && !isValidPosition(newX, newY + 1, currentPiece.state)) {
                locking = true;
                lockTimer = 0;
            }
        }
        return true;
    }
    return false;
}

function rotatePiece(direction) { // 1 = CW, -1 = CCW (or 3)
    if (!currentPiece) return;
    
    const oldState = currentPiece.state;
    let newState;
    if (direction === 1) {
        newState = (oldState + 1) % 4;
    } else {
        newState = (oldState + 3) % 4;
    }
    
    const kickKey = oldState + '>' + newState;
    const kicks = currentPiece.type === 'I' ? KICKS_I : 
                  currentPiece.type === 'O' ? [[0,0]] :
                  KICKS_JLSTZ;
    
    const kickTable = currentPiece.type === 'O' ? [[0,0]] : kicks[kickKey];
    
    for (let i = 0; i < kickTable.length; i++) {
        const [kx, ky] = kickTable[i];
        const newX = currentPiece.x + kx;
        const newY = currentPiece.y + ky;
        
        if (isValidPosition(newX, newY, newState)) {
            currentPiece.x = newX;
            currentPiece.y = newY;
            currentPiece.state = newState;
            lastMoveWasRotation = true;
            currentPiece.lastKick = i;
            
            if (locking) {
                lockTimer = 0;
            }
            // Check if we should start locking
            if (!isValidPosition(newX, newY + 1, newState)) {
                locking = true;
                lockTimer = 0;
            }
            
            playSound('rotate');
            return true;
        }
    }
    return false;
}

function hardDrop() {
    if (!currentPiece) return;
    let dropped = 0;
    while (isValidPosition(currentPiece.x, currentPiece.y + 1, currentPiece.state)) {
        currentPiece.y++;
        dropped++;
    }
    score += dropped * 2;
    playSound('drop');
    lockPiece();
}

function softDropMove() {
    if (!currentPiece) return;
    if (movePiece(0, 1)) {
        score += 1;
        dropTimer = 0;
    }
}

// ==================== T-SPIN DETECTION ====================
function detectTSpin() {
    if (currentPiece.type !== 'T' || !lastMoveWasRotation) return 0;
    
    const x = currentPiece.x;
    const y = currentPiece.y;
    
    // Check 4 corners of the 3x3 bounding box
    const corners = [
        [0, 0], [2, 0], [0, 2], [2, 2]
    ];
    
    let filled = 0;
    for (const [cx, cy] of corners) {
        const bx = x + cx;
        const by = y + cy;
        if (bx < 0 || bx >= COLS || by >= ROWS || 
            (by >= 0 && board[by][bx] !== null)) {
            filled++;
        }
    }
    
    if (filled >= 3) {
        // Check if full T-spin (4 corners) vs mini (3 corners)
        return filled === 4 ? 2 : 1; // 2 = full, 1 = mini
    }
    return 0;
}

// ==================== LINE CLEARING ====================
function lockPiece() {
    if (!currentPiece) return;
    
    const tSpin = detectTSpin();
    const cells = PIECES[currentPiece.type].states[currentPiece.state];
    
    // Lock cells to board
    for (const [cx, cy] of cells) {
        const bx = currentPiece.x + cx;
        const by = currentPiece.y + cy;
        if (by >= 0 && by < ROWS && bx >= 0 && bx < COLS) {
            board[by][bx] = {
                color: PIECES[currentPiece.type].color,
                glow: PIECES[currentPiece.type].glow
            };
        }
    }
    
    // Check for line clears
    clearingRows = [];
    for (let r = 0; r < ROWS; r++) {
        if (board[r].every(cell => cell !== null)) {
            clearingRows.push(r);
        }
    }
    
    if (clearingRows.length > 0) {
        lineClearing = true;
        lineClearTimer = 0;
        processLines(clearingRows.length, tSpin);
        spawnLineClearParticles(clearingRows);
    } else {
        combo = -1;
        currentPiece = null;
        spawnPiece();
    }
    
    locking = false;
    lockTimer = 0;
}

function processLines(count, tSpin) {
    // Calculate score
    let baseScore = 0;
    let linesCleared = count;
    let isSpecial = false;
    let label = '';
    
    if (tSpin > 0) {
        isSpecial = true;
        switch (count) {
            case 0:
                baseScore = tSpin === 2 ? 400 : 100;
                label = (tSpin === 2 ? 'T-SPIN' : 'T-SPIN MINI');
                break;
            case 1:
                baseScore = tSpin === 2 ? 800 : 400;
                label = (tSpin === 2 ? 'T-SPIN SINGLE' : 'T-SPIN MINI SINGLE');
                break;
            case 2:
                baseScore = tSpin === 2 ? 1200 : 600;
                label = (tSpin === 2 ? 'T-SPIN DOUBLE' : 'T-SPIN MINI DOUBLE');
                break;
            case 3:
                baseScore = 1600;
                label = 'T-SPIN TRIPLE';
                break;
        }
    } else {
        switch (count) {
            case 1: baseScore = 100; label = 'SINGLE'; break;
            case 2: baseScore = 300; label = 'DOUBLE'; break;
            case 3: baseScore = 500; label = 'TRIPLE'; break;
            case 4: baseScore = 800; label = 'TETRIS!'; isSpecial = true; break;
        }
    }
    
    // Back-to-back bonus
    if (isSpecial && backToBack) {
        baseScore = Math.floor(baseScore * 1.5);
    }
    if (isSpecial) backToBack = true;
    else if (count > 0) backToBack = false;
    
    // Combo bonus
    combo++;
    if (combo > 0) {
        baseScore += 50 * combo;
    }
    
    // Apply level multiplier
    let earned = baseScore * (level + 1);
    score += earned;
    
    // Update lines and level
    lines += linesCleared;
    level = Math.floor(lines / 10);
    
    // Score popup
    scorePopups.push({
        text: label,
        sub: '+' + earned,
        x: CANVAS_WIDTH / 2,
        y: 300,
        alpha: 1,
        vy: -1
    });
}

function finishLineClear() {
    // Remove cleared rows
    clearingRows.sort((a, b) => b - a);
    for (const row of clearingRows) {
        board.splice(row, 1);
        board.unshift(Array(COLS).fill(null));
    }
    
    clearingRows = [];
    lineClearing = false;
    lineClearTimer = 0;
    currentPiece = null;
    spawnPiece();
}

// ==================== GHOST PIECE ====================
function getGhostY() {
    if (!currentPiece) return 0;
    let ghostY = currentPiece.y;
    while (isValidPosition(currentPiece.x, ghostY + 1, currentPiece.state)) {
        ghostY++;
    }
    return ghostY;
}

// ==================== PARTICLES ====================
function spawnLineClearParticles(rows) {
    for (const row of rows) {
        for (let col = 0; col < COLS; col++) {
            const color = board[row][col]?.color || '#fff';
            const px = col * CELL + CELL / 2;
            const py = row * CELL + CELL / 2;
            for (let i = 0; i < 8; i++) {
                particles.push({
                    x: px,
                    y: py,
                    vx: (Math.random() - 0.5) * 8,
                    vy: (Math.random() - 0.5) * 8 - 2,
                    life: 1,
                    color: color,
                    size: Math.random() * 4 + 2
                });
            }
        }
    }
}

function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15;
        p.life -= 0.02;
        if (p.life <= 0) {
            particles.splice(i, 1);
        }
    }
    
    for (let i = scorePopups.length - 1; i >= 0; i--) {
        const sp = scorePopups[i];
        sp.y += sp.vy;
        sp.alpha -= 0.015;
        if (sp.alpha <= 0) {
            scorePopups.splice(i, 1);
        }
    }
}

let audioCtx = null;

function initAudio() {
    if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }

function playSound(type) {
    if (!audioCtx) return;
    
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    
    switch (type) {
        case 'move':
            osc.frequency.value = 200;
            gain.gain.value = 0.1;
            osc.type = 'square';
            osc.start();
            osc.stop(audioCtx.currentTime + 0.05);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.05);
            break;
        case 'rotate':
            osc.frequency.value = 400;
            gain.gain.value = 0.1;
            osc.type = 'sine';
            osc.start();
            osc.stop(audioCtx.currentTime + 0.1);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            break;
        case 'drop':
            osc.frequency.value = 150;
            gain.gain.value = 0.2;
            osc.type = 'square';
            osc.start();
            osc.stop(audioCtx.currentTime + 0.15);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.15);
            break;
        case 'line':
            osc.frequency.value = 500;
            gain.gain.value = 0.15;
            osc.type = 'sine';
            osc.frequency.setValueAtTime(500, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(1000, audioCtx.currentTime + 0.3);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.3);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            break;
        case 'tetris':
            // Tetris! sound - ascending arpeggio
            [523, 659, 784, 1047].forEach((freq, i) => {
                const o = audioCtx.createOscillator();
                const g = audioCtx.createGain();
                o.connect(g);
                g.connect(audioCtx.destination);
                o.frequency.value = freq;
                o.type = 'sine';
                g.gain.value = 0.1;
                o.start(audioCtx.currentTime + i * 0.08);
                o.stop(audioCtx.currentTime + i * 0.08 + 0.15);
                g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + i * 0.08 + 0.15);
            });
            return;
        case 'hold':
            osc.frequency.value = 300;
            gain.gain.value = 0.1;
            osc.type = 'triangle';
            osc.start();
            osc.stop(audioCtx.currentTime + 0.1);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            break;
        case 'gameover':
            osc.frequency.setValueAtTime(440, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 1);
            gain.gain.value = 0.2;
            osc.type = 'sawtooth';
            osc.start();
            osc.stop(audioCtx.currentTime + 1);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1);
            break;
    }
}

// Background music (simple loop)
let musicPlaying = false;
let musicNodes = [];

function startMusic() {
    if (!audioCtx || musicPlaying) return;
    musicPlaying = true;
    
    // Simple bass line
    const playNote = (freq, time, duration) => {
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        const filter = audioCtx.createBiquadFilter();
        
        osc.connect(filter);
        filter.connect(gain);
        gain.connect(audioCtx.destination);
        
        osc.frequency.value = freq;
        osc.type = 'sine';
        filter.type = 'lowpass';
        filter.frequency.value = 1000;
        gain.gain.setValueAtTime(0.08, time);
        gain.gain.exponentialRampToValueAtTime(0.001, time + duration);
        
        osc.start(time);
        osc.stop(time + duration);
    };
    
    // Korobeiniki melody (simplified)
    const melody = [
        // [frequency, duration in beats]
        [659, 0.5], [494, 0.25], [440, 0.25], [494, 0.5],
        [587, 0.5], [440, 0.25], [370, 0.25], [440, 0.5],
        // ... etc (simplified)
    ];
    
    let t = audioCtx.currentTime;
    const bpm = 140;
    const beatTime = 60 / bpm;
    
    // Play a simple loop
    for (let i = 0; i < melody.length; i++) {
        const [freq, beats] = melody[i];
        playNote(freq, t + i * beatTime, beats * beatTime * 0.9);
    }
}

// ==================== INPUT HANDLING ====================
document.addEventListener('keydown', (e) => {
    if (gameOver) return;
    
    if (e.key === 'p' || e.key === 'P' || e.key === 'Escape') {
        togglePause();
        return;
    }
    
    if (paused) return;
    
    initAudio();
    
    switch (e.key) {
        case 'ArrowLeft':
            if (!keys['ArrowLeft']) {
                movePiece(-1, 0);
                keys['ArrowLeft'] = true;
                dasTimer = 0;
                dasDirection = -1;
                arrTimer = 0;
            }
            e.preventDefault();
            break;
        case 'ArrowRight':
            if (!keys['ArrowRight']) {
                movePiece(1, 0);
                keys['ArrowRight'] = true;
                dasTimer = 0;
                dasDirection = 1;
                arrTimer = 0;
            }
            e.preventDefault();
            break;
        case 'ArrowDown':
            softDrop = true;
            e.preventDefault();
            break;
        case 'ArrowUp':
        case 'x':
        case 'X':
            rotatePiece(1); // CW
            e.preventDefault();
            break;
        case 'z':
        case 'Z':
        case 'Control':
            rotatePiece(-1); // CCW
            e.preventDefault();
            break;
        case ' ':
            hardDrop();
            e.preventDefault();
            break;
        case 'c':
        case 'C':
        case 'Shift':
            holdPiece();
            e.preventDefault();
            break;
    }
});

document.addEventListener('keyup', (e) => {
    switch (e.key) {
        case 'ArrowLeft':
            keys['ArrowLeft'] = false;
            if (dasDirection === -1) dasDirection = 0;
            break;
        case 'ArrowRight':
            keys['ArrowRight'] = false;
            if (dasDirection === 1) dasDirection = 0;
            break;
        case 'ArrowDown':
            softDrop = false;
            break;
    }
});

// DAS handling in game loop
function handleDAS(deltaTime) {
    if (dasDirection !== 0) {
        dasTimer += deltaTime;
        if (dasTimer >= DAS_DELAY) {
            arrTimer += deltaTime;
            if (arrTimer >= ARR_DELAY) {
                arrTimer -= ARR_DELAY;
                movePiece(dasDirection, 0);
            }
        }
    }
}

function holdPiece() {
    if (!currentPiece || holdUsed) return;
    
    const currentType = currentPiece.type;
    
    if (holdPiece) {
        // Swap
        const heldType = holdPiece;
        holdPiece = currentType;
        const piece = PIECES[heldType];
        const boxSize = piece.boxSize;
        currentPiece = {
            type: heldType,
            state: 0,
            x: 3,
            y: -1,
            lastAction: null,
            lastKick: 0
        };
        holdUsed = true;
        locking = false;
        lockTimer = 0;
        dropTimer = 0;
        lastMoveWasRotation = false;
    } else {
        holdPiece = currentType;
        currentPiece = null;
        spawnPiece();
        holdUsed = true;
    }
    
    playSound('hold');
}

// ==================== GAME LOOP ====================
function update(deltaTime) {
    if (gameOver || paused || !currentPiece && !lineClearing) {
        updateParticles();
        return;
    }
    
    if (lineClearing) {
        lineClearTimer += deltaTime;
        if (lineClearTimer >= 400) { // 400ms for line clear animation
            finishLineClear();
        }
        updateParticles();
        return;
    }
    
    if (!currentPiece) {
        updateParticles();
        return;
    }
    
    gameTime += deltaTime;
    
    // Handle DAS
    handleDAS(deltaTime);
    
    // Soft drop
    if (softDrop) {
        dropTimer += deltaTime * 10; // 10x speed
    } else {
        dropTimer += deltaTime;
    }
    
    // Gravity
    const speed = SPEEDS[Math.min(level, SPEEDS.length - 1)];
    if (dropTimer >= speed) {
        dropTimer -= speed;
        if (!movePiece(0, 1)) {
            // Can't move down - start/continue lock delay
            if (!locking) {
                locking = true;
                lockTimer = 0;
            }
        }
    }
    
    // Lock delay
    if (locking && isValidPosition(currentPiece.x, currentPiece.y + 1, currentPiece.state) === false) {
        lockTimer += deltaTime;
        if (lockTimer >= lockDelay) {
            lockPiece();
        }
    } else {
        locking = false;
        lockTimer = 0;
    }
    
    updateParticles();
}

function render() {
    const canvas = document.getElementById('main-canvas');
    const ctx = canvas.getContext('2d');
    
    // Clear
    ctx.clearRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
    
    // Background
    const bgGrad = ctx.createLinearGradient(0, 0, 0, CANVAS_HEIGHT);
    bgGrad.addColorStop(0, '#0a0a2e');
    bgGrad.addColorStop(1, '#111144');
    ctx.fillStyle = bgGrad;
    ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
    
    // Grid lines
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
    ctx.lineWidth = 1;
    for (let r = 0; r <= ROWS; r++) {
        ctx.beginPath();
        ctx.moveTo(0, r * CELL);
        ctx.lineTo(CANVAS_WIDTH, r * CELL);
        ctx.stroke();
    }
    for (let c = 0; c <= COLS; c++) {
        ctx.beginPath();
        ctx.moveTo(c * CELL, 0);
        ctx.lineTo(c * CELL, CANVAS_HEIGHT);
        ctx.stroke();
    }
    
    // Board pieces
    for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {

function drawMiniPiece(ctx, type, x, y, cellSize) {
    const piece = PIECES[type];
    const cells = piece.states[0]; // Always show in state 0
    const size = piece.boxSize;
    
    // Center the piece in the preview
    const offsetX = x + (90 - size * cellSize) / 2; // 90 = canvas width
    const offsetY = y + cellSize;
    
    for (const [cx, cy] of cells) {
        drawCell(ctx, offsetX + cx * cellSize, offsetY + cy * cellSize, 
                piece.color, piece.glow, 1, cellSize);
    }
}

                        const t = audioCtx.currentTime + i * 0.1;
                        g.gain.setValueAtTime(0.08, t);
                        g.gain.exponentialRampToValueAtTime(0.001, t + 0.3);
                        o.start(t); o.stop(t + 0.3);
                    });
                    break;
                }
                case 'gameover': {
                    const o = audioCtx.createOscillator();
                    const g = audioCtx.createGain();
                    o.connect(g); g.connect(audioCtx.destination);
                    o.frequency.setValueAtTime(440, audioCtx.currentTime);
                    o.frequency.exponentialRampToValueAtTime(80, audioCtx.currentTime + 1.5);
                    o.type = 'sawtooth';
                    g.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.5);
                    o.start(); o.stop(audioCtx.currentTime + 1.5);
                    break;
                }
            }
        } catch (e) { /* ignore audio errors */ }
    }
    
    // Background music - Korobeiniki melody
    const NOTE_FREQS = {
        'E5': 659.25, 'D5': 587.33, 'C5': 523.25, 'B4': 493.88,
        'A4': 440.00, 'A5': 880.00, 'G#5': 830.61, 'G5': 783.99,

    function startBGM() {
        if (!audioCtx || musicPlaying) return;
        musicPlaying = true;
        playBGMLoop();
    }
    
    function stopBGM() {
        musicPlaying = false;
        if (bgmLoopTimeout) {
            clearTimeout(bgmLoopTimeout);
            bgmLoopTimeout = null;
        }
        // Stop all scheduled nodes
        for (const node of bgmNodes) {
            try { node.stop(); } catch(e) {}
        }
        bgmNodes = [];
    }
    
    function playBGMLoop() {
        if (!musicPlaying || !audioCtx) return;
        
        const bpm = 130;
        const beat = 60 / bpm;
        const startTime = audioCtx.currentTime + 0.1;
        
        // Korobeiniki melody (Type A)
        const melody = [
            // Section A
            ['E5', 0.5], ['B4', 0.25], ['C5', 0.25], ['D5', 0.5], ['C5', 0.25], ['B4', 0.25],
            ['A4', 0.5], ['A4', 0.25], ['C5', 0.25], ['E5', 0.5], ['D5', 0.25], ['C5', 0.25],
            ['B4', 0.75], ['C5', 0.25], ['D5', 0.5], ['E5', 0.5],
            ['C5', 0.5], ['A4', 0.5], ['A4', 1.0], ['_', 0.5],
            // Section B
            ['E5', 0.5], ['C5', 0.25], ['D5', 0.25], ['B4', 0.5], ['C5', 0.25], ['E5', 0.25],
            ['A5', 0.75], ['G#5', 0.25], ['A5', 0.5], ['G#5', 0.25], ['A5', 0.25],
            ['B4', 0.5], ['G#5', 0.5], ['E5', 1.0],
            // Section A repeat
            ['B4', 0.5], ['C5', 0.25], ['D5', 0.5], ['E5', 0.5],
            ['C5', 0.5], ['A4', 0.5], ['A4', 1.0],
            // Section C
            ['E5', 0.5], ['C5', 0.25], ['D5', 0.25], ['B4', 0.5], ['C5', 0.25], ['E5', 0.25],
            ['A5', 0.75], ['G#5', 0.25], ['A5', 0.5], ['G#5', 0.25], ['A5', 0.25],
            ['B4', 0.5], ['G#5', 0.5], ['E5', 0.5],
            ['C5', 0.5], ['D5', 0.5], ['B4', 0.5], ['C5', 1.0],
            ['D5', 1.0], ['B4', 0.5], ['E5', 1.0],
            ['E5', 1.5], ['_', 0.5],
        ];
        
        // Bass line (simplified)
        const bass = [
            // Simplified: one note per bar
            ['A3', 1.0], ['E3', 1.0], ['A3', 1.0], ['E3', 1.0],
            ['A3', 1.0], ['E3', 1.0], ['G3', 1.0], ['C4', 1.0],

        ];
        
        let t = startTime;
        let totalDuration = 0;
        
        for (const [note, dur] of melody) {
            const duration = dur * beat;
            if (note !== '_' && NOTE_FREQS[note]) {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                const filter = audioCtx.createBiquadFilter();
                osc.connect(filter);
                filter.connect(gain);
                gain.connect(audioCtx.destination);
                
                osc.frequency.value = NOTE_FREQS[note];
                osc.type = 'square';
                filter.type = 'lowpass';
                filter.frequency.value = 2000;
                gain.gain.setValueAtTime(0.035, t);
                gain.gain.exponentialRampToValueAtTime(0.001, t + duration * 0.95);
                
                osc.start(t);
                osc.stop(t + duration);
                bgmNodes.push(osc);
            }
            t += duration;
            totalDuration += duration;
        }
        
        // Play bass
        let bt = startTime;
        const bassNoteDur = beat;
        for (const [note, dur] of bass) {
            const duration = dur * bassNoteDur;
            if (note !== '_' && NOTE_FREQS[note]) {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain); gain.connect(audioCtx.destination);
                osc.frequency.value = NOTE_FREQS[note];
                osc.type = 'triangle';
                gain.gain.setValueAtTime(0.06, bt);
                gain.gain.exponentialRampToValueAtTime(0.001, bt + duration * 0.9);
                osc.start(bt);
                osc.stop(bt + duration);
                bgmNodes.push(osc);
            }
            bt += duration;
        }
        
        // Schedule next loop
        bgmLoopTimeout = setTimeout(() => {
            bgmNodes = [];
            if (musicPlaying) playBGMLoop();
        }, (totalDuration + 0.2) * 1000);
    }
    
    // ============ GAME STATE ============
    let board, currentPiece, holdType, holdUsed;
    let nextQueue, score, level, lines, combo, backToBack;
    let gameOver, paused, gameStarted;
    let dropTimer, lockTimer, locking;
    let lineClearing, lineClearTimer, clearingRows;
    let lastMoveWasRotation;
    let particles, scorePopups, screenShake;
    let gameTime, lastFrameTime;
    let keys, dasTimer, dasDirection, arrTimer, softDrop;
    
    const DAS_DELAY = 160;
    const ARR_DELAY = 45;
    const LOCK_DELAY = 500;
    const LINE_CLEAR_TIME = 400;
    
    // Background particles
    let bgParticles = [];
    
    // ============ INITIALIZATION ============
    function initGame() {
        board = Array.from({length: ROWS}, () => Array(COLS).fill(null));
        currentPiece = null;
        holdType = null;
        holdUsed = false;
        nextQueue = [];
        score = 0;
        level = 0;
        lines = 0;
        combo = -1;
        backToBack = false;
        gameOver = false;
        paused = false;
        dropTimer = 0;
        lockTimer = 0;
        locking = false;
        lineClearing = false;
        lineClearTimer = 0;
        clearingRows = [];
        lastMoveWasRotation = false;
        particles = [];
        scorePopups = [];
        screenShake = {x: 0, y: 0, intensity: 0};
        gameTime = 0;
        keys = {};
        dasTimer = 0;
        dasDirection = 0;
        arrTimer = 0;
        softDrop = false;
        
        refillQueue();
        spawnPiece();
        updateStats();
    }
    
    function refillQueue() {
        const bag = ['I','O','T','S','Z','J','L'];
        // Fisher-Yates shuffle
        for (let i = bag.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [bag[i], bag[j]] = [bag[j], bag[i]];
        }
        nextQueue.push(...bag);
    }
    
    function spawnPiece() {
        if (nextQueue.length < 7) refillQueue();
        const type = nextQueue.shift();
        const def = PIECES[type];
        
        currentPiece = {
            type,
            state: 0,
            x: def.boxSize === 4 ? 3 : 3,
            y: -1,
            lastKick: 0
        };
        
        holdUsed = false;
        lastMoveWasRotation = false;
        locking = false;
        lockTimer = 0;
        dropTimer = 0;
        
        // Check spawn collision
        if (!isValidPos(currentPiece.x, currentPiece.y, 0, type)) {
            gameOver = true;
            stopBGM();
            playSound('gameover');
            showOverlay('gameover-overlay');
            updateFinalStats();
        }
    }
    
    // ============ COLLISION ============
    function isValidPos(x, y, state, type) {
        const cells = PIECES[type].states[state];
        for (const [cx, cy] of cells) {
            const bx = x + cx;
            const by = y + cy;
            if (bx < 0 || bx >= COLS) return false;
            if (by >= ROWS) return false;
            if (by >= 0 && board[by][bx] !== null) return false;
        }
        return true;
    }
    
    function isValidCurrent(x, y, state) {
        return isValidPos(x, y, state, currentPiece.type);
    }
    
    // ============ MOVEMENT ============
    function tryMove(dx, dy) {
        if (!currentPiece) return false;
        const nx = currentPiece.x + dx;
        const ny = currentPiece.y + dy;
        if (isValidCurrent(nx, ny, currentPiece.state)) {
            currentPiece.x = nx;
            currentPiece.y = ny;
            lastMoveWasRotation = false;
            if (locking) lockTimer = 0;
            if (dy > 0 && !isValidCurrent(nx, ny + 1, currentPiece.state)) {
                locking = true;
            }
            return true;
        }
        return false;
    }
    
    function tryRotate(dir) {
        if (!currentPiece) return false;
        const oldState = currentPiece.state;
        const newState = (oldState + (dir === 1 ? 1 : 3)) % 4;
        const kickKey = oldState + '>' + newState;
        
        let kicks;
        if (currentPiece.type === 'I') kicks = KICKS_I[kickKey];
        else if (currentPiece.type === 'O') kicks = [[0,0]];
        else kicks = KICKS_JLSTZ[kickKey];
        
        if (!kicks) return false;
        
        for (let i = 0; i < kicks.length; i++) {
            const [kx, ky] = kicks[i];
            const nx = currentPiece.x + kx;
            const ny = currentPiece.y + ky;
            if (isValidCurrent(nx, ny, newState)) {
                currentPiece.x = nx;
                currentPiece.y = ny;
                currentPiece.state = newState;
                currentPiece.lastKick = i;
                lastMoveWasRotation = true;
                if (locking) lockTimer = 0;
                if (!isValidCurrent(nx, ny + 1, newState)) {
                    locking = true;
                }
                playSound('rotate');
                return true;
            }
        }
        return false;
    }
    
    function hardDrop() {
        if (!currentPiece) return;
        let dropped = 0;
        while (isValidCurrent(currentPiece.x, currentPiece.y + 1, currentPiece.state)) {
            currentPiece.y++;
            dropped++;
        }
        score += dropped * 2;
        playSound('drop');
        lockPiece();
    }
    
    function holdCurrent() {
        if (!currentPiece || holdUsed) return;
        const currentType = currentPiece.type;
        
        if (holdType) {
            const swap = holdType;
            holdType = currentType;
            const def = PIECES[swap];
            currentPiece = { type: swap, state: 0, x: 3, y: -1, lastKick: 0 };
            if (!isValidCurrent(currentPiece.x, currentPiece.y, 0)) {
                gameOver = true;
                stopBGM();
                playSound('gameover');
                showOverlay('gameover-overlay');
                updateFinalStats();
                return;
            }
        } else {
            holdType = currentType;
            spawnPiece();
        }
        
        holdUsed = true;
        locking = false;
        lockTimer = 0;
        dropTimer = 0;
        lastMoveWasRotation = false;
        playSound('hold');
    }
    
    // ============ T-SPIN DETECTION ============
    function detectTSpin() {
        if (currentPiece.type !== 'T' || !lastMoveWasRotation) return 0;
        
        const x = currentPiece.x;
        const y = currentPiece.y;
        const corners = [[0,0],[2,0],[0,2],[2,2]];
        let filled = 0;
        
        for (const [cx, cy] of corners) {
            const bx = x + cx;
            const by = y + cy;
            if (bx < 0 || bx >= COLS || by >= ROWS || (by >= 0 && board[by][bx] !== null)) {
                filled++;
            }
        }
        
        return filled >= 4 ? 2 : filled >= 3 ? 1 : 0;
    }
    
    // ============ LOCKING & LINE CLEARS ============
    function lockPiece() {
        if (!currentPiece) return;
        
        const tSpin = detectTSpin();
        const cells = PIECES[currentPiece.type].states[currentPiece.state];
        const def = PIECES[currentPiece.type];
        
        for (const [cx, cy] of cells) {
            const bx = currentPiece.x + cx;
            const by = currentPiece.y + cy;
            if (by >= 0 && by < ROWS && bx >= 0 && bx < COLS) {
                board[by][bx] = { color: def.color, glow: def.glow };
            }
        }
        
        playSound('lock');
        
        // Check lines
        clearingRows = [];
        for (let r = 0; r < ROWS; r++) {
            if (board[r].every(cell => cell !== null)) {
                clearingRows.push(r);
            }
        }
        
        if (clearingRows.length > 0) {
            lineClearing = true;
            lineClearTimer = 0;
            calculateScore(clearingRows.length, tSpin);
            spawnLineParticles(clearingRows);
            screenShake.intensity = clearingRows.length >= 4 ? 8 : clearingRows.length * 2;
        } else {
            combo = -1;
            if (tSpin > 0) {
                // T-spin with no lines
                const tsScore = (tSpin === 2 ? 400 : 100) * (level + 1);
                score += tsScore;
                addPopup(tSpin === 2 ? 'T-SPIN' : 'T-SPIN MINI', '+' + tsScore);
            }
            currentPiece = null;
            spawnPiece();
        }
        
        locking = false;
        lockTimer = 0;
    }
    
    function calculateScore(cleared, tSpin) {
        let base = 0;
        let label = '';
        let special = false;
        
        if (tSpin > 0) {
            special = true;
            const names = ['', ['T-SPIN MINI SINGLE','T-SPIN SINGLE','T-SPIN DOUBLE','T-SPIN TRIPLE'],
                                ['T-SPIN SINGLE','T-SPIN SINGLE','T-SPIN DOUBLE','T-SPIN TRIPLE']];
            if (tSpin === 1) {
                base = [100,400,600,800][cleared];
                label = ['', 'T-SPIN MINI S', 'T-SPIN MINI D', 'T-SPIN TRIPLE'][cleared];
            } else {
                base = [400,800,1200,1600][cleared];
                label = ['', 'T-SPIN S', 'T-SPIN D', 'T-SPIN TRIPLE'][cleared];
            }
            // For 3 lines with T-spin (regardless mini/full), it's always triple
            if (cleared === 3) {
                base = 1600;
                label = 'T-SPIN TRIPLE!';
            }
        } else {
            const scores = [0, 100, 300, 500, 800];
            const labels = ['', 'SINGLE', 'DOUBLE', 'TRIPLE', 'TETRIS!'];
            base = scores[cleared];
            label = labels[cleared];
            if (cleared === 4) special = true;
        }
        
        // Back-to-back
        if (special && backToBack) {
            base = Math.floor(base * 1.5);
            label += ' B2B';
        }
        if (special) backToBack = true;
        else backToBack = false;
        
        // Combo
        combo++;
        if (combo > 0) {
            base += 50 * combo;
            label += ' x' + combo;
        }
        
        // Level multiplier
        const earned = base * (level + 1);
        score += earned;
        
        // Level up check
        const oldLevel = level;
        lines += cleared;
        level = Math.floor(lines / 10);
        
        if (level > oldLevel) {
            playSound('levelup');
            addPopup('LEVEL ' + level, '');
        }
        
        // Sounds
        if (tSpin > 0) playSound('tspin');
        else if (cleared === 4) playSound('tetris');
        else playSound('line');
        
        // Popup
        addPopup(label, '+' + earned);
        updateStats();
    }
    
    function finishLineClear() {
        // Remove rows (process from bottom to top so indices stay valid)
        clearingRows.sort((a, b) => a - b);
        for (let i = clearingRows.length - 1; i >= 0; i--) {
            board.splice(clearingRows[i], 1);
        }
        // Add empty rows at top
        for (let i = 0; i < clearingRows.length; i++) {
            board.unshift(Array(COLS).fill(null));
        }
        
        clearingRows = [];
        lineClearing = false;
        lineClearTimer = 0;
        currentPiece = null;
        spawnPiece();
        updateStats();
    }
    
    // ============ GHOST ============
    function getGhostY() {
        if (!currentPiece) return 0;
        let gy = currentPiece.y;
        while (isValidCurrent(currentPiece.x, gy + 1, currentPiece.state)) gy++;
        return gy;
    }
    
    // ============ PARTICLES & EFFECTS ============
    function spawnLineParticles(rows) {
        for (const row of rows) {
            for (let col = 0; col < COLS; col++) {
                const cell = board[row][col];
                if (!cell) continue;
                const px = col * CELL + CELL / 2;
                const py = row * CELL + CELL / 2;
                for (let i = 0; i < 6; i++) {
                    particles.push({
                        x: px + (Math.random() - 0.5) * CELL * 0.8,
                        y: py + (Math.random() - 0.5) * CELL * 0.8,
                        vx: (Math.random() - 0.5) * 6,
                        vy: (Math.random() - 0.5) * 6 - 2,
                        life: 1,
                        decay: 0.015 + Math.random() * 0.01,
                        color: cell.color,
                        size: 2 + Math.random() * 5
                    });
                }
            }
        }
    }
    
    function addPopup(text, sub) {
        scorePopups.push({
            text, sub,
            x: W / 2,
            y: H / 2 - 50,
            alpha: 1,
            vy: -1.5
        });
    }
    
    function updateEffects() {
        for (let i = particles.length - 1; i >= 0; i--) {
            const p = particles[i];
            p.x += p.vx;
            p.y += p.vy;
            p.vy += 0.12;
            p.life -= p.decay;
            if (p.life <= 0) particles.splice(i, 1);
        }
        
        for (let i = scorePopups.length - 1; i >= 0; i--) {
            const s = scorePopups[i];
            s.y += s.vy;
            s.alpha -= 0.012;
            if (s.alpha <= 0) scorePopups.splice(i, 1);
        }
        
        // Screen shake
        if (screenShake.intensity > 0) {
            screenShake.x = (Math.random() - 0.5) * screenShake.intensity;
            screenShake.y = (Math.random() - 0.5) * screenShake.intensity;
            screenShake.intensity *= 0.9;
            if (screenShake.intensity < 0.5) {
                screenShake.intensity = 0;
                screenShake.x = 0;
                screenShake.y = 0;
            }
        }
        
        // Background particles
        for (const bp of bgParticles) {
            bp.y -= bp.speed;
            bp.x += Math.sin(bp.y * 0.01) * 0.3;
            if (bp.y < -10) {
                bp.y = H + 10;
                bp.x = Math.random() * W;
            }
        }
    }
    
    function initBgParticles() {
        bgParticles = [];
        for (let i = 0; i < 30; i++) {
            bgParticles.push({
                x: Math.random() * W,
                y: Math.random() * H,
                size: Math.random() * 2 + 0.5,
                speed: Math.random() * 0.5 + 0.2,
                alpha: Math.random() * 0.3 + 0.1
            });
        }
    }
    
    // ============ INPUT ============
    document.addEventListener('keydown', (e) => {
        if (gameOver) return;
        if (e.key === 'p' || e.key === 'P' || e.key === 'Escape') {
            togglePause();
            e.preventDefault();
            return;
        }
        if (paused || !gameStarted) return;
        
        initAudio();
        
        switch (e.key) {
            case 'ArrowLeft':
                if (!keys['ArrowLeft']) {
                    tryMove(-1, 0);
                    keys['ArrowLeft'] = true;
                    dasTimer = 0; dasDirection = -1; arrTimer = 0;
                }
                e.preventDefault(); break;
            case 'ArrowRight':
                if (!keys['ArrowRight']) {
                    tryMove(1, 0);
                    keys['ArrowRight'] = true;
                    dasTimer = 0; dasDirection = 1; arrTimer = 0;
                }
                e.preventDefault(); break;
            case 'ArrowDown':
                softDrop = true;
                e.preventDefault(); break;
            case 'ArrowUp': case 'x': case 'X':
                tryRotate(1);
                e.preventDefault(); break;
            case 'z': case 'Z': case 'Control':
                tryRotate(-1);
                e.preventDefault(); break;
            case ' ':
                hardDrop();
                e.preventDefault(); break;
            case 'c': case 'C': case 'Shift':
                holdCurrent();
                e.preventDefault(); break;
        }
    });
    
    document.addEventListener('keyup', (e) => {
        switch (e.key) {
            case 'ArrowLeft':
                keys['ArrowLeft'] = false;
                if (dasDirection === -1) dasDirection = 0;
                break;
            case 'ArrowRight':
                keys['ArrowRight'] = false;
                if (dasDirection === 1) dasDirection = 0;
                break;
            case 'ArrowDown':
                softDrop = false;
                break;
        }
    });
    
    function handleDAS(dt) {
        if (dasDirection === 0) return;
        dasTimer += dt;
        if (dasTimer >= DAS_DELAY) {
            arrTimer += dt;
            while (arrTimer >= ARR_DELAY) {
                arrTimer -= ARR_DELAY;
                tryMove(dasDirection, 0);
            }
        }
    }
    
    // Mobile controls
    function setupMobileControls() {
        document.querySelectorAll('.touch-btn').forEach(btn => {
            const action = btn.dataset.action;
            
            btn.addEventListener('touchstart', (e) => {
                e.preventDefault();
                if (gameOver || paused || !gameStarted) return;
                initAudio();
                
                switch (action) {
                    case 'left': tryMove(-1, 0); dasDirection = -1; dasTimer = 0; break;
                    case 'right': tryMove(1, 0); dasDirection = 1; dasTimer = 0; break;
                    case 'down': softDrop = true; break;
                    case 'rotate': tryRotate(1); break;
                    case 'ccw': tryRotate(-1); break;
                    case 'hard': hardDrop(); break;
                    case 'hold': holdCurrent(); break;
                    case 'pause': togglePause(); break;
                }
            });
            
            btn.addEventListener('touchend', (e) => {
                e.preventDefault();
                switch (action) {
                    case 'down': softDrop = false; break;
                    case 'left': if (dasDirection === -1) dasDirection = 0; break;
                    case 'right': if (dasDirection === 1) dasDirection = 0; break;
                }
            });
        });
    }
    
    // Touch swipe on play field
    function setupTouchSwipe() {
        const canvas = document.getElementById('main-canvas');
        let touchStartX = 0, touchStartY = 0, touchStartTime = 0;
        let swipeHandled = false;
        
        canvas.addEventListener('touchstart', (e) => {
            e.preventDefault();
            initAudio();
            const touch = e.touches[0];
            touchStartX = touch.clientX;
            touchStartY = touch.clientY;
            touchStartTime = Date.now();
            swipeHandled = false;
        });
        
        canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            if (swipeHandled) return;
            
            const touch = e.touches[0];
            const dx = touch.clientX - touchStartX;
            const dy = touch.clientY - touchStartY;
            const cellPx = CELL * (canvas.width / 300); // scaled

    function drawMiniPiece(ctx, type, ox, oy, maxW, maxH, alpha, scale) {
