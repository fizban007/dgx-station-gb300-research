```javascript
function drawCell(ctx, x, y, size, color, alpha = 1) {
  ctx.save();
  ctx.globalAlpha = alpha;
  
  // Outer glow
  ctx.shadowColor = color;
  ctx.shadowBlur = 15;
  
  // Gradient fill
  const gradient = ctx.createLinearGradient(x, y, x + size, y + size);
  gradient.addColorStop(0, lightenColor(color, 40));
  gradient.addColorStop(0.5, color);
  gradient.addColorStop(1, darkenColor(color, 30));
  
  ctx.fillStyle = gradient;
  ctx.fillRect(x + 1, y + 1, size - 2, size - 2);
  
  // Inner highlight
  ctx.shadowBlur = 0;
  const innerGrad = ctx.createLinearGradient(x, y, x, y + size * 0.4);
  innerGrad.addColorStop(0, 'rgba(255,255,255,0.3)');
  innerGrad.addColorStop(1, 'rgba(255,255,255,0)');
  ctx.fillStyle = innerGrad;
  ctx.fillRect(x + 1, y + 1, size - 2, size * 0.4);
  
  ctx.restore();
}
```

```javascript
class AudioManager {
  constructor() {
    this.ctx = null;
    this.initialized = false;
  }
  
  init() {
    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
    this.initialized = true;
  }
  
  playNote(freq, duration, type = 'square', volume = 0.1) {
    if (!this.initialized) return;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    gain.gain.setValueAtTime(volume, this.ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
    osc.connect(gain);
    gain.connect(this.ctx.destination);
    osc.start();
    osc.stop(this.ctx.currentTime + duration);
  }
  
  playMove() { this.playNote(440, 0.05, 'square', 0.05); }
  playRotate() { this.playNote(660, 0.05, 'square', 0.05); }
  playDrop() { this.playNote(110, 0.1, 'triangle', 0.1); }
  playLock() { this.playNote(220, 0.05, 'triangle', 0.08); }

}
```

```javascript
function drawCell(ctx, x, y, size, color, opts = {}) {
  const { alpha = 1, ghost = false, glow = true, inner = true } = opts;
  
  ctx.save();
  ctx.globalAlpha = alpha;
  
  if (ghost) {
    // Draw ghost cell
    ctx.strokeStyle = color;
    ctx.lineWidth = 2;
    ctx.globalAlpha = 0.3;
    ctx.strokeRect(x + 1, y + 1, size - 2, size - 2);
  } else {
    // Draw filled cell
    if (glow) {
      ctx.shadowColor = color;
      ctx.shadowBlur = 10;
    }
    
    // Main gradient
    const grad = ctx.createLinearGradient(x, y, x + size, y + size);
    grad.addColorStop(0, lighten(color, 30));
    grad.addColorStop(0.5, color);
    grad.addColorStop(1, darken(color, 20));
    ctx.fillStyle = grad;
    
    // Draw with slight inset for grid appearance
    ctx.fillRect(x + 0.5, y + 0.5, size - 1, size - 1);
    
    // Inner highlight
    if (inner) {
      ctx.shadowBlur = 0;
      const hGrad = ctx.createLinearGradient(x, y, x, y + size * 0.3);
      hGrad.addColorStop(0, 'rgba(255,255,255,0.4)');
      hGrad.addColorStop(1, 'rgba(255,255,255,0)');
      ctx.fillStyle = hGrad;
      ctx.fillRect(x + 1, y + 1, size - 2, size * 0.3);
      
      // Bottom shadow
      const sGrad = ctx.createLinearGradient(x, y + size * 0.7, x, y + size);
      sGrad.addColorStop(0, 'rgba(0,0,0,0)');
      sGrad.addColorStop(1, 'rgba(0,0,0,0.3)');
      ctx.fillStyle = sGrad;
      ctx.fillRect(x + 1, y + size * 0.7, size - 2, size * 0.3 - 1);
    }
  }
  
  ctx.restore();
}
```

```javascript
function calculateScore(linesCleared, tspinType, level, combo, b2b) {
  let score = 0;
  let isDifficult = false;
  
  if (tspinType === 'regular') {
    // T-spin
    isDifficult = true;
    if (linesCleared === 0) score = 400;
    else if (linesCleared === 1) score = 800;
    else if (linesCleared === 2) score = 1200;
    else if (linesCleared === 3) score = 1600;
  } else if (tspinType === 'mini') {
    // Mini T-spin
    isDifficult = true;
    if (linesCleared === 0) score = 100;
    else if (linesCleared === 1) score = 200;
    else if (linesCleared === 2) score = 400;
    else if (linesCleared === 3) score = 400; // unlikely but possible
  } else {
    // Normal line clear
    if (linesCleared === 1) score = 100;
    else if (linesCleared === 2) score = 300;
    else if (linesCleared === 3) score = 500;
    else if (linesCleared === 4) {
      score = 800;
      isDifficult = true;
    }
  }
  
  score *= level;
  
  // B2B bonus
  if (b2b > 0 && isDifficult) {
    score = Math.floor(score * 1.5);
  }
  
  // Combo bonus
  if (combo > 0) {
    score += 50 * combo * level;
  }
  
  return score;
}
```

```javascript
checkTSpin() {
  if (this.current.type !== 'T' || !this.lastActionWasRotation) return null;
  
  // Check 4 corners of the T piece bounding box
  const px = this.current.x;
  const py = this.current.y;
  const corners = [
    [px, py],           // top-left
    [px + 2, py],       // top-right
    [px, py + 2],       // bottom-left
    [px + 2, py + 2],   // bottom-right
  ];
  
  let filledCorners = 0;
  const cornerStates = corners.map(([cx, cy]) => {
    const filled = this.isFilled(cx, cy);
    if (filled) filledCorners++;
    return filled;
  });
  
  if (filledCorners < 3) return null;
  
  // Check front corners to determine T-spin vs Mini T-spin
  // Front depends on T piece rotation state
  const state = this.current.rotation;
  let frontIndices;
  switch (state) {
    case 0: frontIndices = [0, 1]; break; // top-left, top-right
    case 1: frontIndices = [1, 3]; break; // top-right, bottom-right
    case 2: frontIndices = [2, 3]; break; // bottom-left, bottom-right
    case 3: frontIndices = [0, 2]; break; // top-left, bottom-left
  }
  
  const bothFront = cornerStates[frontIndices[0]] && cornerStates[frontIndices[1]];
  
  return bothFront ? 'regular' : 'mini';
}

}
```

```javascript
function drawPreviewPiece(ctx, pieceType, cx, cy, cellSize) {
  const piece = PIECES[pieceType];
  const matrix = piece.matrices[0];
  const color = piece.color;
  
  // Calculate bounding box of filled cells
  let minX = 10, maxX = -1, minY = 10, maxY = -1;
  for (let r = 0; r < matrix.length; r++) {
    for (let c = 0; c < matrix[r].length; c++) {
      if (matrix[r][c]) {
        minX = Math.min(minX, c);
        maxX = Math.max(maxX, c);
        minY = Math.min(minY, r);
        maxY = Math.max(maxY, r);
      }
    }
  }
  
  const pieceW = (maxX - minX + 1) * cellSize;
  const pieceH = (maxY - minY + 1) * cellSize;
  const startX = cx - pieceW / 2;
  const startY = cy - pieceH / 2;
  
  for (let r = minY; r <= maxY; r++) {
    for (let c = minX; c <= maxX; c++) {
      if (matrix[r][c]) {
        drawCell(ctx, startX + (c - minX) * cellSize, startY + (r - minY) * cellSize, cellSize, color);
      }
    }
  }
```

```javascript
  ctx.save();
  ctx.globalAlpha = alpha;

  hg.addColorStop(1, 'rgba(255,255,255,0)');
  ctx.fillStyle = hg;
  roundRect(ctx, x + 2, y + 2, size - 4, size * 0.35, r);
  ctx.fill();

function roundRect(ctx, x, y, w, h, r) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.lineTo(x + w - r, y);
  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
  ctx.lineTo(x + w, y + h - r);
  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
  ctx.lineTo(x + r, y + h);
  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
  ctx.lineTo(x, y + r);
  ctx.quadraticCurveTo(x, y, x + r, y);
  ctx.closePath();
}

function lightenColor(hex, amount) {
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  return `rgb(${Math.min(255, r + amount)}, ${Math.min(255, g + amount)}, ${Math.min(255, b + amount)})`;
}

function darkenColor(hex, amount) {
  const r = parseInt(hex.slice(1, 3), 16);
  const g = parseInt(hex.slice(3, 5), 16);
  const b = parseInt(hex.slice(5, 7), 16);
  return `rgb(${Math.max(0, r - amount)}, ${Math.max(0, g - amount)}, ${Math.max(0, b - amount)})`;
}
```
