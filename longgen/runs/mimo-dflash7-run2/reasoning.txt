```javascript
        ctx.strokeStyle = color;
        ctx.lineWidth = 2;
```

```javascript
    // Create a simple bass loop using oscillators
    const bassNotes = [82.41, 82.41, 98.00, 98.00, 73.42, 73.42, 65.41, 65.41]; // E2, E2, G2, G2, D2, D2, C2, C2

        if (!musicPlaying || isPaused || isGameOver) return;
        
        const freq = bassNotes[noteIndex % bassNotes.length];
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        
        osc.type = 'sawtooth';
        osc.frequency.value = freq;
        gain.gain.value = 0.15;
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + noteDuration * 0.9);

        gain.connect(audioCtx.destination);

        osc.stop(audioCtx.currentTime + noteDuration);
        
        // Add hi-hat
        if (noteIndex % 2 === 0) {
            createNoise(0.05, 0.1);
        }
        
        noteIndex++;
    }, noteDuration * 1000);
}
```

```javascript
function fillBag() {
    const pieces = ['I', 'O', 'T', 'S', 'Z', 'J', 'L'];
    // Shuffle
    for (let i = pieces.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [pieces[i], pieces[j]] = [pieces[j], pieces[i]];
    }
    bag.push(...pieces);
}

function getNextPiece() {
    if (bag.length < 7) fillBag();
    return bag.shift();
}
```

This ensures each group of 7 pieces contains all 7 types.

```javascript
function removeLines(fullRows) {
    const rowSet = new Set(fullRows);
    const newBoard = [];
    
    for (let r = 0; r < ROWS; r++) {
        if (!rowSet.has(r)) {
            newBoard.push(board[r]);

    for (let i = 0; i < rowsRemoved; i++) {
```

```javascript
const particles = [];

function spawnLineParticles(row) {
    for (let x = 0; x < COLS; x++) {
        const color = board[row][x] || '#ffffff';
        for (let i = 0; i < 5; i++) {
            particles.push({
                x: x * CELL + CELL / 2,
                y: row * CELL + CELL / 2,
                vx: (Math.random() - 0.5) * 6,
                vy: (Math.random() - 0.5) * 4 - 3,
                color: color,
                life: 1,
                decay: 0.015 + Math.random() * 0.02,
                size: 2 + Math.random() * 4,
            });
        }
    }
}

function updateParticles(dt) {
    const factor = dt / 16; // normalize to 60fps
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * factor;
        p.y += p.vy * factor;
        p.vy += 0.1 * factor; // gravity
        p.life -= p.decay * factor;
        
        if (p.life <= 0) {
            particles.splice(i, 1);
        }
    }
}

function drawParticles(ctx) {
    ctx.globalCompositeOperation = 'lighter';
    for (const p of particles) {
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.color;
        ctx.shadowColor = p.color;
        ctx.shadowBlur = p.size * 2;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx.fill();
    }
    ctx.globalCompositeOperation = 'source-over';
    ctx.globalAlpha = 1;
}
```

```javascript
class AudioEngine {
    constructor() {
        this.ctx = null;
        this.musicPlaying = false;
        this.musicTimer = null;
        this.muted = false;
    }
    
    init() {
        if (this.ctx) return;
        this.ctx = new (window.AudioContext || window.webkitAudioContext)();
    }
    
    playTone(freq, duration, type = 'square', volume = 0.3) {
        if (!this.ctx || this.muted) return;
        
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        
        osc.type = type;
        osc.frequency.value = freq;
        gain.gain.value = volume;
        gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
        
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + duration);
    }
    
    playNoise(duration, volume = 0.2) {
        if (!this.ctx || this.muted) return;
        
        const bufferSize = this.ctx.sampleRate * duration;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);
        
        for (let i = 0; i < bufferSize; i++) {
            data[i] = Math.random() * 2 - 1;
        }
        
        const source = this.ctx.createBufferSource();
        const gain = this.ctx.createGain();
        
        source.buffer = buffer;
        gain.gain.value = volume;
        gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
        
        source.connect(gain);
        gain.connect(this.ctx.destination);
        source.start();
    }
    
    move() {
        this.playTone(200, 0.05, 'square', 0.15);
    }
    
    rotate() {
        this.playTone(300, 0.08, 'square', 0.2);
    }
    
    hardDrop() {

    softDrop() {

            this.musicTimer = null;
        }
    }
    
    toggleMute() {
        this.muted = !this.muted;
        return this.muted;
```

```javascript
    if (ghost) {
        ctx.globalAlpha = 0.25;
        ctx.strokeStyle = color;
        ctx.lineWidth = 2;
        ctx.setLineDash([4, 4]);

function adjustBrightness(hex, amount) {
    const num = parseInt(hex.replace('#', ''), 16);
    const r = Math.min(255, Math.max(0, (num >> 16) + amount));
    const g = Math.min(255, Math.max(0, ((num >> 8) & 0xFF) + amount));
    const b = Math.min(255, Math.max(0, (num & 0xFF) + amount));
    return `rgb(${r},${g},${b})`;
}
```
